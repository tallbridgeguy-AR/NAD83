// Bulk fetch ALL elevations in a SINGLE fast POST request async function fetchBulkElevations(rawPoints) { const url = "https://api.open-elevation.com/api/v1/lookup"; // Format all coordinates into the bulk JSON payload const payload = { locations: rawPoints.map(p => ({ latitude: p.lat, longitude: p.lon })) }; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!response.ok) { throw new Error(`Elevation API server returned status ${response.status}`); } const data = await response.json(); // Extract elevation array from response if (data && data.results) { return data.results.map(r => r.elevation || 0); } return new Array(rawPoints.length).fill(0); }
